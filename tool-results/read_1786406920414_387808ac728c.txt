   174→function autoMapColumns(csvHeaders: string[], entityType: EntityType): Record<string, string> {
   175→  const mapping: Record<string, string> = {}
   176→  const fields = ENTITY_FIELDS[entityType]
   177→
   178→  for (const field of fields) {
   179→    const patterns = AUTO_MAP_PATTERNS[field.value] || []
   180→    for (const header of csvHeaders) {
   181→      const normalized = header.toLowerCase().replace(/\s+/g, '_').replace(/-/g, '_')
   182→      if (patterns.some(p => p === normalized || p.replace(/_/g, ' ') === header.toLowerCase())) {
   183→        mapping[field.value] = header
   184→        break
   185→      }
   186→    }
   187→  }
   188→  return mapping
   189→}

   191→function formatBytes(bytes: number): string {
   192→  if (bytes < 1024) return `${bytes} B`
   193→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   194→  return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   195→}
   196→
   197→function formatDate(dateStr: string | null): string {
   198→  if (!dateStr) return '—'
   199→  return new Date(dateStr).toLocaleDateString('en-US', {
   200→    month: 'short',
   201→    day: 'numeric',
   202→    year: 'numeric',
   203→    hour: '2-digit',
   204→    minute: '2-digit',
   205→  })
   206→}

   218→}

   220→function parseCSVHeaders(text: string): string[] {
   221→  const firstLine = text.split(/\r?\n/)[0] || ''
   222→  const headers: string[] = []
   223→  let inQuotes = false
   224→  let current = ''
   225→  for (const ch of firstLine) {
   226→    if (inQuotes) {
   227→      if (ch === '"' && current.endsWith('"')) {
   228→        inQuotes = false
   229→      } else {
   230→        current += ch
   231→      }
   232→    } else {
   233→      if (ch === '"') {
   234→        inQuotes = true
   235→      } else if (ch === ',') {
   236→        headers.push(current.trim())
   237→        current = ''
   238→      } else {
   239→        current += ch
   240→      }
   241→    }
   242→  }
   243→  if (current.trim()) headers.push(current.trim())
   244→  return headers
   245→}

   249→export default function MigrationView() {

   255→  const [showMapping, setShowMapping] = useState(false)
   256→  const [isImporting, setIsImporting] = useState(false)
   257→  const [importProgress, setImportProgress] = useState(0)
   258→  const [importResult, setImportResult] = useState<MigrationRecord | null>(null)
   259→  const [importErrors, setImportErrors] = useState<ImportError[]>([])
   260→
   261→  // ── State: History ──
   262→  const [migrations, setMigrations] = useState<MigrationRecord[]>([])
   263→  const [historyLoading, setHistoryLoading] = useState(true)
   264→  const [rollbackId, setRollbackId] = useState<string | null>(null)
   265→  const [isRollingBack, setIsRollingBack] = useState(false)
   266→
   267→  // Drag state
   268→  const [isDragOver, setIsDragOver] = useState(false)
   269→  const fileInputRef = useRef<HTMLInputElement>(null)
   270→
   271→  // ── Fetch migration history ──
   272→  const fetchHistory = useCallback(async () => {
   273→    try {
   274→      const data = await migrationApi.list()
   275→      setMigrations(data.data || [])
   276→    } catch (err) {
   277→      toast.error(err instanceof Error ? err.message : 'Failed to load migration history')
   278→    } finally {
   279→      setHistoryLoading(false)
   280→    }
   281→  }, [])
   282→
   283→  useEffect(() => {
   284→    fetchHistory()
   285→  }, [fetchHistory])
   286→
   287→  // ── Process uploaded file ──
   288→  const processFile = useCallback((selectedFile: File) => {
   289→    if (!selectedFile.name.endsWith('.csv')) {
   290→      toast.error('Only CSV files are supported')
   291→      return
   292→    }
   293→    setFile(selectedFile)
   294→    setImportResult(null)
   295→    setImportErrors([])
   296→
   297→    const reader = new FileReader()
   298→    reader.onload = (e) => {
   299→      const text = e.target?.result as string
   300→      const headers = parseCSVHeaders(text)
   301→      setCsvHeaders(headers)
   302→      const mapping = autoMapColumns(headers, entityType)
   303→      setColumnMapping(mapping)
   304→      setShowMapping(true)
   305→    }
   306→    reader.onerror = () => {
   307→      toast.error('Failed to read file')
   308→    }
   309→    reader.readAsText(selectedFile)
   310→  }, [entityType])
   311→
   312→  // ── Handle entity type change ──
   313→  const handleEntityTypeChange = (value: string) => {
   314→    const type = value as EntityType
   315→    setEntityType(type)
   316→    setImportResult(null)
   317→    setImportErrors([])
   318→    if (file) {
   319→      const reader = new FileReader()
   320→      reader.onload = (e) => {
   321→        const text = e.target?.result as string
   322→        const headers = parseCSVHeaders(text)
   323→        setCsvHeaders(headers)
   324→        const mapping = autoMapColumns(headers, type)
   325→        setColumnMapping(mapping)
   326→      }
   327→      reader.readAsText(file)
   328→    } else {
   329→      setCsvHeaders([])
   330→      setColumnMapping({})
   331→      setShowMapping(false)
   332→    }
   333→  }
   334→
   335→  // ── Drag & drop handlers ──
   336→  const handleDragOver = (e: React.DragEvent) => {
   337→    e.preventDefault()
   338→    setIsDragOver(true)
   339→  }
   340→
   341→  const handleDragLeave = (e: React.DragEvent) => {
   342→    e.preventDefault()
   343→    setIsDragOver(false)
   344→  }
   345→
   346→  const handleDrop = (e: React.DragEvent) => {
   347→    e.preventDefault()
   348→    setIsDragOver(false)
   349→    const droppedFile = e.dataTransfer.files[0]
   350→    if (droppedFile) processFile(droppedFile)
   351→  }
   352→
   353→  const handleFileChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   354→    const selectedFile = e.target.files?.[0]
   355→    if (selectedFile) processFile(selectedFile)
   356→  }
   357→
   358→  // ── Update mapping ──
   359→  const updateMapping = (targetField: string, sourceColumn: string) => {
   360→    setColumnMapping((prev) => {
   361→      const next = { ...prev }
   362→      // Remove any other mapping pointing to the same source column
   363→      for (const [key, val] of Object.entries(next)) {
   364→        if (val === sourceColumn && key !== targetField) {
   365→          delete next[key]
   366→        }
   367→      }
   368→      if (sourceColumn === '__none__') {
   369→        delete next[targetField]
   370→      } else {
   371→        next[targetField] = sourceColumn
   372→      }
   373→      return next
   374→    })
   375→  }
   376→
   377→  // ── Start import ──
   378→  const handleImport = async () => {
   379→    if (!file) {
   380→      toast.error('Please select a file first')
   381→      return
   382→    }
   383→
   384→    setIsImporting(true)
   385→    setImportProgress(10)
   386→    setImportResult(null)
   387→    setImportErrors([])
   388→
   389→    try {
   390→      const formData = new FormData()
   391→      formData.append('file', file)
   392→      formData.append('entityType', entityType)
   393→      formData.append('mappingConfig', JSON.stringify(columnMapping))
   394→
   395→      setImportProgress(30)
   396→      const data = await migrationApi.importCSV(formData)
   397→      setImportProgress(100)
   398→
   399→      const migration = data.data as MigrationRecord
   400→      setImportResult(migration)
   401→
   402→      // Parse errors if any
   403→      if (migration.errors) {
   404→        try {
   405→          const parsed = JSON.parse(migration.errors)
   406→          setImportErrors(Array.isArray(parsed) ? parsed : [])
   407→        } catch {
   408→          setImportErrors([])
   409→        }
   410→      }
   411→
   412→      if (migration.status === 'completed') {
   413→        toast.success(`Import completed: ${migration.processedRows} rows processed successfully`)
   414→      } else if (migration.status === 'failed') {
   415→        toast.error('Import failed. Check the error details below.')
   416→      } else {
   417→        toast.info(`Import finished with status: ${migration.status}`)
   418→      }
   419→
   420→      // Refresh history
   421→      fetchHistory()
   422→    } catch (err) {
   423→      setImportProgress(0)
   424→      toast.error(err instanceof Error ? err.message : 'Import failed')
   425→    } finally {
   426→      setIsImporting(false)
   427→    }
   428→  }
   429→
   430→  // ── Rollback ──
   431→  const handleRollback = async () => {
   432→    if (!rollbackId) return
   433→    setIsRollingBack(true)

   435→      const data = await migrationApi.rollback(rollbackId)
   436→      toast.success(data.message || 'Migration rolled back successfully')
   437→      setRollbackId(null)
   438→      fetchHistory()
   439→    } catch (err) {
   440→      toast.error(err instanceof Error ? err.message : 'Rollback failed')
   441→    } finally {
   442→      setIsRollingBack(false)
   443→    }
   444→  }
   445→
   446→  // ── Reset import form ──
   447→  const resetForm = () => {
   448→    setFile(null)
   449→    setCsvHeaders([])
   450→    setColumnMapping({})
   451→    setShowMapping(false)
   452→    setImportResult(null)
   453→    setImportErrors([])
   454→    setImportProgress(0)
   455→    if (fileInputRef.current) fileInputRef.current.value = ''
   456→  }
   457→
   458→  // ── Render ──
   459→  return (
   460→    <div className="flex h-full flex-col gap-6 overflow-y-auto p-4 md:p-6">
   461→      {/* Header */}
   462→      <div>
   463→        <h1 className="text-xl font-semibold text-foreground">Data Migration</h1>
   464→        <p className="mt-1 text-sm text-muted-foreground">
   465→          Import data from CSV files into your AssetHub system. Map columns, review, and track import history.
   466→        </p>
   467→      </div>
   468→
   469→      {/* Section 1: Import Data */}
   470→      <Card>
   471→        <CardHeader className="pb-4">
   472→          <div className="flex items-center gap-2">
   473→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-[#0f766e]/10">
   474→              <Upload className="h-4 w-4 text-[#0f766e]" />
   475→            </div>
   476→            <div>
   477→              <CardTitle className="text-base">Import Data</CardTitle>
   478→              <CardDescription>Upload a CSV file and map columns to target fields</CardDescription>
   479→            </div>
   480→          </div>
   481→        </CardHeader>
   482→        <CardContent className="space-y-5">
   483→          {/* Entity type selector */}
   484→          <div className="space-y-2">
   485→            <Label className="text-sm font-medium">Entity Type</Label>
   486→            <Select value={entityType} onValueChange={handleEntityTypeChange}>
   487→              <SelectTrigger className="w-full sm:w-64">
   488→                <SelectValue />
   489→              </SelectTrigger>
   490→              <SelectContent>
   491→                {(Object.keys(ENTITY_FIELDS) as EntityType[]).map((type) => (
   492→                  <SelectItem key={type} value={type}>
   493→                    <div className="flex items-center gap-2">
   494→                      {type === 'assets' && <FileSpreadsheet className="h-3.5 w-3.5 text-muted-foreground" />}
   495→                      {type === 'categories' && <Database className="h-3.5 w-3.5 text-muted-foreground" />}
   496→                      {type === 'locations' && <Database className="h-3.5 w-3.5 text-muted-foreground" />}
   497→                      {type === 'users' && <Database className="h-3.5 w-3.5 text-muted-foreground" />}
   498→                      {ENTITY_LABELS[type]}
   499→                    </div>
   500→                  </SelectItem>
   501→                ))}
   502→              </SelectContent>
   503→            </Select>
   504→          </div>
   505→
   506→          {/* Upload area */}
   507→          <div className="space-y-2">
   508→            <Label className="text-sm font-medium">CSV File</Label>
   509→            <div
   510→              onDragOver={handleDragOver}
   511→              onDragLeave={handleDragLeave}
   512→              onDrop={handleDrop}
   513→              onClick={() => fileInputRef.current?.click()}
   514→              className={`relative flex cursor-pointer flex-col items-center justify-center gap-3 rounded-xl border-2 border-dashed p-8 transition-colors ${
   515→                isDragOver
   516→                  ? 'border-[#0f766e] bg-[#0f766e]/5'
   517→                  : file
   518→                    ? 'border-green-300 bg-green-50/50'
   519→                    : 'border-slate-200 bg-slate-50/50 hover:border-slate-300 hover:bg-slate-50'

   522→              <input
   523→                ref={fileInputRef}
   524→                type="file"

   526→                className="hidden"
   527→                onChange={handleFileChange}
   528→              />

   532→                    <CheckCircle2 className="h-6 w-6 text-green-600" />
   533→                  </div>

   535→                    <p className="text-sm font-medium text-foreground">{file.name}</p>
   536→                    <p className="text-xs text-muted-foreground">{formatBytes(file.size)}</p>

   541→                  >

   559→          </div>
   560→
   561→          {/* Column mapping */}

   569→              >

   571→                  <div className="mb-3 flex items-center justify-between">
   572→                    <div className="flex items-center gap-2">
   573→                      <ArrowRight className="h-4 w-4 text-[#0f766e]" />
   574→                      <span className="text-sm font-medium text-foreground">Column Mapping</span>

   581→                  <ScrollArea className="max-h-64">

   597→                            <SelectContent>
   598→                              <SelectItem value="__none__">— Not mapped —</SelectItem>

   601→                                  {header}
   602→                                </SelectItem>
   603→                              ))}
   604→                            </SelectContent>
   605→                          </Select>
   606→                        </div>
   607→                      ))}
   608→                    </div>
   609→                  </ScrollArea>
   610→                </div>
   611→              </motion.div>
   612→            )}
   613→          </AnimatePresence>
   614→
   615→          {/* Import progress */}
   616→          {isImporting && (
   617→            <motion.div
   618→              initial={{ opacity: 0, y: 8 }}
   619→              animate={{ opacity: 1, y: 0 }}
   620→              className="space-y-3 rounded-xl border border-amber-200 bg-amber-50 p-4"
   621→            >
   622→              <div className="flex items-center gap-2">
   623→                <Loader2 className="h-4 w-4 animate-spin text-amber-600" />
   624→                <span className="text-sm font-medium text-amber-800">Importing data...</span>
   625→              </div>
   626→              <Progress value={importProgress} className="h-2 bg-amber-100" />
   627→              <p className="text-xs text-amber-600">Please wait while your data is being processed.</p>
   628→            </motion.div>
   629→          )}
   630→
   631→          {/* Import results */}
   632→          <AnimatePresence>
   633→            {importResult && !isImporting && (
   634→              <motion.div
   635→                initial={{ opacity: 0, y: 8 }}
   636→                animate={{ opacity: 1, y: 0 }}
   637→                exit={{ opacity: 0, y: -8 }}
   638→              >
   639→                <div className={`rounded-xl border p-4 ${
   640→                  importResult.status === 'completed'
   641→                    ? 'border-green-200 bg-green-50'
   642→                    : 'border-red-200 bg-red-50'
   643→                }`}>
   644→                  <div className="flex items-start gap-3">
   645→                    {importResult.status === 'completed' ? (
   646→                      <CheckCircle2 className="mt-0.5 h-5 w-5 shrink-0 text-green-600" />
   647→                    ) : (
   648→                      <XCircle className="mt-0.5 h-5 w-5 shrink-0 text-red-500" />

   650→                    <div className="flex-1">
   651→                      <h3 className={`text-sm font-semibold ${
   652→                        importResult.status === 'completed' ? 'text-green-800' : 'text-red-800'
   653→                      }`}>
   654→                        {importResult.status === 'completed' ? 'Import Completed' : 'Import Issues Found'}
   655→                      </h3>
   656→                      <div className="mt-2 flex flex-wrap gap-4 text-sm">
   657→                        <span className={importResult.processedRows > 0 ? 'text-green-700' : 'text-muted-foreground'}>
   658→                          <strong>{importResult.processedRows}</strong> successful
   659→                        </span>
   660→                        <span className={importResult.failedRows > 0 ? 'text-red-600' : 'text-muted-foreground'}>
   661→                          <strong>{importResult.failedRows}</strong> failed
   662→                        </span>
   663→                        <span className="text-muted-foreground">
   664→                          <strong>{importResult.totalRows}</strong> total
   665→                        </span>
   666→                      </div>
   667→
   668→                      {/* Error list */}
   669→                      {importErrors.length > 0 && (
   670→                        <div className="mt-3">

   672→                            onClick={() => {
   673→                              const el = document.getElementById('import-errors')
   674→                              el?.classList.toggle('hidden')

   676→                            className="text-xs font-medium text-red-600 hover:text-red-700"

   678→                            {importErrors.length} error{importErrors.length > 1 ? 's' : ''} — click to {document.getElementById('import-errors')?.classList.contains('hidden') ? 'show' : 'hide'}

   681→                            <ScrollArea className="max-h-48 rounded-lg border border-red-200 bg-white">
   682→                              <div className="space-y-1 p-2">
   683→                                {importErrors.slice(0, 50).map((err, idx) => (
   684→                                  <div key={idx} className="flex items-start gap-2 rounded px-2 py-1 text-xs">
   685→                                    <span className="shrink-0 font-mono text-red-400">Row {err.row}:</span>
   686→                                    <span className="text-red-700">{err.message}</span>
   687→                                  </div>
   688→                                ))}
   689→                                {importErrors.length > 50 && (
   690→                                  <p className="px-2 py-1 text-xs text-muted-foreground">
   691→                                    ...and {importErrors.length - 50} more errors
   692→                                  </p>
   693→                                )}
   694→                              </div>
   695→                            </ScrollArea>
   696→                          </div>
   697→                        </div>
   698→                      )}
   699→                    </div>
   700→                  </div>
   701→                </div>
   702→              </motion.div>
   703→            )}
   704→          </AnimatePresence>
   705→
   706→          {/* Import button */}
   707→          <div className="flex items-center gap-3">
   708→            <Button
   709→              className="bg-[#0f766e] text-white hover:bg-[#0d6560]"
   710→              disabled={!file || isImporting}
   711→              onClick={handleImport}
   712→            >
   713→              {isImporting ? (
   714→                <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   715→              ) : (
   716→                <Play className="mr-2 h-4 w-4" />
   717→              )}
   718→              {isImporting ? 'Importing...' : 'Start Import'}
   719→            </Button>
   720→            {file && !isImporting && (
   721→              <Button variant="ghost" onClick={resetForm}>
   722→                <Trash2 className="mr-2 h-4 w-4" />
   723→                Clear
   724→              </Button>
   725→            )}
   726→          </div>
   727→        </CardContent>
   728→      </Card>
   729→
   730→      {/* Section 2: Import History */}
   731→      <Card>
   732→        <CardHeader className="pb-4">
   733→          <div className="flex items-center gap-2">
   734→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-[#0f766e]/10">
   735→              <History className="h-4 w-4 text-[#0f766e]" />
   736→            </div>
   737→            <div>
   738→              <CardTitle className="text-base">Import History</CardTitle>
   739→              <CardDescription>View and manage past data imports</CardDescription>
   740→            </div>
   741→          </div>
   742→        </CardHeader>
   743→        <CardContent>
   744→          {historyLoading ? (
   745→            <div className="space-y-3">
   746→              {Array.from({ length: 4 }).map((_, i) => (
   747→                <Skeleton key={i} className="h-12 w-full rounded-lg" />
   748→              ))}
   749→            </div>
   750→          ) : migrations.length === 0 ? (
   751→            <div className="flex flex-col items-center justify-center gap-2 py-12 text-center">
   752→              <Database className="h-10 w-10 text-slate-300" />
   753→              <p className="text-sm text-muted-foreground">No import history yet</p>
   754→              <p className="text-xs text-muted-foreground">

   756→              </p>
   757→            </div>
   758→          ) : (
   759→            <div className="rounded-lg border">
   760→              <div className="overflow-x-auto">
   761→                <Table>
   762→                  <TableHeader>
   763→                    <TableRow className="bg-slate-50/80">
   764→                      <TableHead className="text-xs font-medium">File Name</TableHead>
   765→                      <TableHead className="text-xs font-medium">Entity</TableHead>
   766→                      <TableHead className="text-xs font-medium">Status</TableHead>
   767→                      <TableHead className="text-xs font-medium text-right">Rows</TableHead>
   768→                      <TableHead className="hidden text-xs font-medium sm:table-cell">Date</TableHead>
   769→                      <TableHead className="text-right text-xs font-medium">Actions</TableHead>
   770→                    </TableRow>
   771→                  </TableHeader>
   772→                  <TableBody>
   773→                    {migrations.map((m) => (
   774→                      <TableRow key={m.id} className="group">
   775→                        <TableCell>
   776→                          <div className="flex items-center gap-2">
   777→                            <FileSpreadsheet className="h-4 w-4 shrink-0 text-slate-400" />

   782→                          </div>
   783→                        </TableCell>
   784→                        <TableCell>
   785→                          <Badge variant="outline" className="capitalize">
   786→                            {m.entityType}
   787→                          </Badge>
   788→                        </TableCell>
   789→                        <TableCell>{statusBadge(m.status)}</TableCell>
   790→                        <TableCell className="text-right text-sm">
   791→                          <span className={m.failedRows > 0 ? 'text-red-600' : ''}>
   792→                            {m.processedRows}
   793→                          </span>
   794→                          <span className="text-muted-foreground">/{m.totalRows}</span>

   796→                        <TableCell className="hidden text-sm text-muted-foreground sm:table-cell">

   799→                        <TableCell className="text-right">
   800→                          {m.status === 'completed' && (
   801→                            <Button
   802→                              variant="ghost"
   803→                              size="sm"
   804→                              className="text-red-600 hover:bg-red-50 hover:text-red-700"
   805→                              onClick={() => setRollbackId(m.id)}
   806→                            >
   807→                              <RotateCcw className="mr-1.5 h-3.5 w-3.5" />
   808→                              <span className="hidden sm:inline">Rollback</span>
   809→                            </Button>
   810→                          )}
   811→                        </TableCell>
   812→                      </TableRow>
   813→                    ))}
   814→                  </TableBody>
   815→                </Table>
   816→              </div>
   817→            </div>
   818→          )}
   819→        </CardContent>
   820→      </Card>
   821→
   822→      {/* Rollback confirmation dialog */}
   823→      <AlertDialog open={!!rollbackId} onOpenChange={(open) => !open && setRollbackId(null)}>
   824→        <AlertDialogContent>

   829→              records may still exist in the system. You may need to manually clean up the data.

   831→          </AlertDialogHeader>
   832→          <AlertDialogFooter>
   833→            <AlertDialogCancel disabled={isRollingBack}>Cancel</AlertDialogCancel>
   834→            <AlertDialogAction
   835→              onClick={handleRollback}
   836→              disabled={isRollingBack}
   837→              className="bg-red-600 text-white hover:bg-red-700"
   838→            >
   839→              {isRollingBack ? (
