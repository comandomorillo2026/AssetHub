    87→    87→function formatTTD(value: number | undefined | null): string {
    88→    88→  if (value == null) return '—'
    89→    89→  return new Intl.NumberFormat('en-TT', {
    90→    90→    style: 'currency',
    91→    91→    currency: 'TTD',
    92→    92→    minimumFractionDigits: 2,
    93→    93→  }).format(value)
    94→    94→}

    96→    96→function formatDate(date: string | undefined | null): string {
    97→    97→  if (!date) return '—'
    98→    98→  return new Date(date).toLocaleDateString('en-US', {
    99→    99→    year: 'numeric',
   100→   100→    month: 'short',
   101→   101→    day: 'numeric',
   102→   102→  })
   103→   103→}
