    70→function relativeTime(dateStr: string): string {
    71→  const now = Date.now()
    72→  const then = new Date(dateStr).getTime()
    73→  const diffMs = now - then
    74→  const seconds = Math.floor(diffMs / 1000)
    75→  const minutes = Math.floor(seconds / 60)
    76→  const hours = Math.floor(minutes / 60)
    77→  const days = Math.floor(hours / 24)
    78→  const weeks = Math.floor(days / 7)
    79→  const months = Math.floor(days / 30)
    80→
    81→  if (seconds < 60) return 'just now'
    82→  if (minutes < 60) return `${minutes}m ago`
    83→  if (hours < 24) return `${hours}h ago`
    84→  if (days < 7) return `${days}d ago`
    85→  if (weeks < 5) return `${weeks}w ago`
    86→  return `${months}mo ago`
    87→}

   181→      try {
   182→        const data = await inventoryApi.list({ status: 'in_progress' })

   186→          setActiveSession(sessions[0])
   187→          setScanCount(sessions[0].totalScanned || 0)
   188→        }
   189→      } catch (_e) {
   190→        // no active session
   191→      }
   192→    }
   193→    checkSession()
   194→    return () => { cancelled = true }
   195→  }, [])
   196→
   197→  // Initialize scanner
   198→  useEffect(() => {
   199→    const scanner = new Html5Qrcode(SCANNER_ID)
   200→    scannerRef.current = scanner
   201→
   202→    scanner
   203→      .start(
   204→        { facingMode: 'environment' },
   205→        { fps: 10, qrbox: { width: 250, height: 250 } },

   215→              try {
   216→                await scanner.resume()

   219→                // ignore resume errors
   220→              }
   221→              setProcessing(false)
   222→            }, 2000)
   223→          }
   224→        },
   225→        () => {}
   226→      )
   227→      .catch(() => {
   228→        // camera not available, that's ok
   229→      })
   230→
   231→    return () => {
   232→      scanner
   233→        .stop()
   234→        .then(() => scanner.clear())
   235→        .catch(() => {})
   236→    }
   237→  }, [])

   249→        }
   250→      }
   251→    } catch (_e) {
   252→      toast.error('Flash not supported on this device')
   253→    }
   254→  }, [flashOn])
   255→
   256→  // Handle scan result
   257→  const handleScanResult = useCallback(async (code: string) => {
   258→    try {
   259→      const data = await qrApi.resolve(code)

   261→      if (asset && asset.id) {
   262→        setScannedAsset(asset)
   263→        setShowAssetCard(true)
   264→        toast.success('Asset found!')
   265→
   266→        // Auto-submit to active inventory session
   267→        if (activeSession) {
   268→          try {
   269→            await inventoryApi.scan(activeSession.id, code)
   270→            setScanCount((prev) => prev + 1)
   271→          } catch (_e) {
   272→            // silently fail for inventory scan
   273→          }
   274→        }
   275→      } else {
   276→        toast.error('Asset not registered')
   277→      }
   278→
   279→      // Add to recent scans
   280→      setRecentScans((prev) => {
   281→        const entry: ScanResult = {
   282→          qrCode: code,
   283→          asset: asset?.id ? asset : null,
   284→          timestamp: Date.now(),
   285→        }
   286→        return [entry, ...prev].slice(0, 5)
   287→      })
   288→    } catch (_e) {
   289→      toast.error('Asset not registered')
   290→      setRecentScans((prev) => {
   291→        const entry: ScanResult = { qrCode: code, asset: null, timestamp: Date.now() }
   292→        return [entry, ...prev].slice(0, 5)
   293→      })
   294→    }
   295→  }, [activeSession])
   296→
   297→  // Manual submit
   298→  const handleManualSubmit = useCallback(async () => {
   299→    const code = manualCode.trim()

   301→    setManualCode('')
   302→    setProcessing(true)
   303→    try {
   304→      if (scannerRef.current) {
   305→        await scannerRef.current.pause(true)
   306→        setScanning(false)
   307→      }
   308→      await handleScanResult(code)
   309→    } finally {
   310→      setTimeout(async () => {
   311→        try {
   312→          if (scannerRef.current) await scannerRef.current.resume()
   313→          setScanning(true)
   314→        } catch (_e) {}
   315→        setProcessing(false)
   316→      }, 2000)
   317→    }
   318→  }, [manualCode, handleScanResult])
   319→
   320→  return (
   321→    <div className="flex flex-col gap-4 p-4 md:p-6 max-w-2xl mx-auto w-full">
   322→      {/* Header */}
   323→      <motion.div {...fadeIn} className="flex items-center justify-between">
   324→        <div className="flex items-center gap-2">
   325→          <ScanLine className="h-5 w-5 text-primary" />
   326→          <h2 className="text-lg font-semibold">QR Scanner</h2>
   327→        </div>
   328→        <Button
   329→          variant="outline"
   330→          size="sm"
   331→          onClick={toggleFlash}
   332→          className="gap-1.5"
   333→        >
   334→          {flashOn ? <ZapOff className="h-4 w-4" /> : <Zap className="h-4 w-4" />}
   335→          {flashOn ? 'Flash Off' : 'Flash'}
   336→        </Button>
   337→      </motion.div>
   338→
   339→      {/* Camera Viewfinder */}

   351→              {/* Scanning line */}

   353→                className="absolute left-1 right-1 h-0.5 bg-gradient-to-r from-transparent via-primary to-transparent"
   354→                initial={{ top: '0%' }}
   355→                animate={{ top: ['0%', '100%', '0%'] }}

   358→            </div>
   359→          </div>

   361→        {/* Not scanning indicator */}
   362→        {!scanning && (
   363→          <div className="absolute inset-0 bg-black/40 flex items-center justify-center">
   364→            <div className="flex flex-col items-center gap-2 text-white">
   365→              <CheckCircle2 className="h-10 w-10 text-emerald-400" />
   366→              <span className="text-sm font-medium">Scan captured</span>

   389→      {/* Start session button if no active session */}
   390→      {!activeSession && (
   391→        <motion.div {...fadeIn}>
   392→          <Button
   393→            className="w-full gap-2"
