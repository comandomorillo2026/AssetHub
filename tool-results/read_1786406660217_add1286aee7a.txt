   140→function formatCurrency(value: number | null): string {
   141→  if (value == null) return '—'
   142→  return `TTD $${value.toLocaleString('en-US', {
   143→    minimumFractionDigits: 2,
   144→    maximumFractionDigits: 2,
   145→  })}`
   146→}
   147→
   148→function formatDate(dateStr: string | null): string {
   149→  if (!dateStr) return '—'
   150→  return new Date(dateStr).toLocaleDateString('en-US', {
   151→    month: 'short',
   152→    day: 'numeric',
   153→    year: 'numeric',
   154→  })
   155→}
   156→
   157→function formatDateTime(dateStr: string | null): string {
   158→  if (!dateStr) return '—'
   159→  return new Date(dateStr).toLocaleDateString('en-US', {
   160→    month: 'short',
   161→    day: 'numeric',
   162→    year: 'numeric',
   163→    hour: '2-digit',
   164→    minute: '2-digit',
   165→  })
   166→}

   180→function statusColor(status: string): string {

   187→  }
   188→}
