   811→                  {session.location.name}
   812→                </span>
   813→              )}

   815→                <Clock className="h-3 w-3" />
   816→                Started {formatDate(session.startedAt)}
   817→              </span>
   818→              {session.user && (
   819→                <span className="flex items-center gap-1">
   820→                  <User className="h-3 w-3" />
   821→                  {session.user.name}
   822→                </span>
   823→              )}
   824→            </div>
   825→          </div>
   826→        </div>
   827→
   828→        <div className="flex gap-2 shrink-0">
   829→          <Button variant="outline" className="gap-1.5" onClick={() => navigate('scan')}>
   830→            <ScanLine className="h-4 w-4" />
   831→            Scan
   832→          </Button>
   833→          {isInProgress && (
   834→            <Button
   835→              className="gap-1.5"
   836→              onClick={() => setShowCompleteDialog(true)}
   837→            >
   838→              <CheckCheck className="h-4 w-4" />
   839→              Complete Session
   840→            </Button>
   841→          )}
   842→        </div>
   843→      </motion.div>
   844→
   845→      {/* Stats Cards */}
   846→      <motion.div
   847→        {...staggerContainer}
   848→        initial="initial"
   849→        animate="animate"
   850→        className="grid grid-cols-2 sm:grid-cols-5 gap-3"
   851→      >
   852→        {[
   853→          {
   854→            label: 'Expected',
   855→            value: session.totalExpected,
   856→            icon: ListChecks,
   857→            color: 'text-slate-600',
   858→            bg: 'bg-slate-50',
   859→          },
   860→          {
   861→            label: 'Scanned',
   862→            value: session.totalScanned,
   863→            icon: ScanLine,
   864→            color: 'text-primary',
   865→            bg: 'bg-primary/10',
   866→          },
   867→          {
   868→            label: 'Matched',
   869→            value: matchedCount,
   870→            icon: CheckCircle2,
   871→            color: 'text-emerald-600',
   872→            bg: 'bg-emerald-50',
   873→          },
   874→          {
   875→            label: 'Missing',
   876→            value: session.totalMissing,
   877→            icon: AlertTriangle,
   878→            color: 'text-amber-600',
   879→            bg: 'bg-amber-50',
   880→          },
   881→          {
   882→            label: 'Extra',
   883→            value: session.totalExtra,
   884→            icon: XCircle,
   885→            color: 'text-red-600',
   886→            bg: 'bg-red-50',
   887→          },
   888→        ].map((stat) => (
   889→          <motion.div key={stat.label} {...fadeIn}>
   890→            <Card className={`${stat.bg} border-0`}>
   891→              <CardContent className="p-3 sm:p-4 flex flex-col items-center text-center gap-1">
   892→                <stat.icon className={`h-5 w-5 ${stat.color}`} />
   893→                <span className="text-xl sm:text-2xl font-bold">{stat.value}</span>
   894→                <span className="text-xs text-muted-foreground">{stat.label}</span>
   895→              </CardContent>
   896→            </Card>
   897→          </motion.div>
   898→        ))}
   899→      </motion.div>
   900→
   901→      {/* Completion Summary */}
   902→      <AnimatePresence>
   903→        {isCompleted && (
   904→          <motion.div
   905→            initial={{ opacity: 0, height: 0 }}
   906→            animate={{ opacity: 1, height: 'auto' }}
   907→            exit={{ opacity: 0, height: 0 }}
   908→          >
   909→            <Card className="border-emerald-200 bg-emerald-50/50">
   910→              <CardContent className="p-4">
   911→                <div className="flex items-center justify-between">
   912→                  <div className="flex items-center gap-2">
   913→                    <CheckCheck className="h-5 w-5 text-emerald-600" />
   914→                    <span className="font-semibold text-emerald-800">
   915→                      Session Completed
   916→                    </span>
   917→                  </div>
   918→                  <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200 hover:bg-emerald-100">
   919→                    {matchRate}% Match Rate
   920→                  </Badge>
   921→                </div>
   922→                <div className="mt-3 space-y-2">
   923→                  <div className="flex items-center justify-between text-sm">
   924→                    <span className="text-emerald-700">Matched items</span>
   925→                    <span className="font-medium">{matchedCount}</span>
   926→                  </div>
   927→                  <div className="flex items-center justify-between text-sm">
   928→                    <span className="text-amber-700">Discrepancies</span>
   929→                    <span className="font-medium">{discrepancyCount}</span>
   930→                  </div>
   931→                  <Progress value={matchRate} className="h-2 mt-1" />
   932→                </div>
   933→                {session.completedAt && (
   934→                  <p className="text-xs text-emerald-600/70 mt-2">
   935→                    Completed on {formatDate(session.completedAt)}
   936→                  </p>
   937→                )}
   938→              </CardContent>
   939→            </Card>
   940→          </motion.div>
   941→        )}
   942→      </AnimatePresence>
   943→
   944→      {/* Scanned Items Table */}
   945→      <motion.div {...fadeIn}>
   946→        <Card>
   947→          <CardHeader className="pb-3">
   948→            <CardTitle className="text-base">Scanned Items</CardTitle>
   949→            <CardDescription>
   950→              {items.length} item{items.length !== 1 ? 's' : ''} scanned
   951→            </CardDescription>
   952→          </CardHeader>
   953→          <CardContent className="p-0">
   954→            {items.length === 0 ? (
   955→              <div className="flex flex-col items-center justify-center py-10 text-center">

   958→              </div>
   959→            ) : (
   960→              <div className="overflow-x-auto max-h-96 overflow-y-auto">
   961→                <Table>
   962→                  <TableHeader>
   963→                    <TableRow>
   964→                      <TableHead className="w-[140px]">QR Code</TableHead>
   965→                      <TableHead>Asset Name</TableHead>
   966→                      <TableHead className="w-[100px]">Status</TableHead>
   967→                      <TableHead className="w-[140px]">Discrepancy</TableHead>
   968→                      <TableHead className="w-[100px]">Time</TableHead>
   969→                    </TableRow>
   970→                  </TableHeader>
   971→                  <TableBody>
   972→                    {items.map((item) => {
   973→                      const hasDiscrepancy = !!item.discrepancyType
   974→                      const rowClass = hasDiscrepancy
   975→                        ? item.discrepancyType === 'missing'
   976→                          ? 'bg-amber-50/80 dark:bg-amber-950/20'
   977→                          : 'bg-red-50/80 dark:bg-red-950/20'
   978→                        : ''
   979→
   980→                      return (
   981→                        <TableRow key={item.id} className={rowClass}>
   982→                          <TableCell className="font-mono text-xs">
   983→                            {item.qrCode || (item.asset?.qrCode ? `...${item.asset.qrCode.slice(-8)}` : '—')}
   984→                          </TableCell>
   985→                          <TableCell className="font-medium text-sm">
   986→                            {item.asset?.name || 'Unregistered'}
   987→                          </TableCell>
   988→                          <TableCell>
   989→                            {item.asset ? (
   990→                              <Badge
   991→                                variant="secondary"
   992→                                className={assetStatusColor(item.asset.status)}
   993→                              >
   994→                                {item.asset.status}
   995→                              </Badge>
   996→                            ) : (
   997→                              <Badge variant="outline">unknown</Badge>
   998→                            )}
   999→                          </TableCell>
  1000→                          <TableCell>
  1001→                            {hasDiscrepancy ? (
  1002→                              <Badge
  1003→                                variant={item.discrepancyType === 'missing' ? 'outline' : 'destructive'}
  1004→                                className={
  1005→                                  item.discrepancyType === 'missing'
  1006→                                    ? 'border-amber-300 text-amber-700 bg-amber-50'
  1007→                                    : ''
  1008→                                }
  1009→                              >
  1010→                                {item.discrepancyType === 'missing' && (
  1011→                                  <AlertTriangle className="h-3 w-3 mr-1" />
  1012→                                )}
  1013→                                {item.discrepancyType === 'extra' && (
  1014→                                  <XCircle className="h-3 w-3 mr-1" />
  1015→                                )}
  1016→                                {item.discrepancyType}
  1017→                                {item.discrepancyNote && (
  1018→                                  <span className="ml-1 text-[10px] opacity-70">
  1019→                                    — {item.discrepancyNote}
  1020→                                  </span>
  1021→                                )}
  1022→                              </Badge>
  1023→                            ) : (
  1024→                              <span className="text-xs text-muted-foreground">—</span>
  1025→                            )}
  1026→                          </TableCell>
  1027→                          <TableCell className="text-xs text-muted-foreground">
  1028→                            {new Date(item.scannedAt).toLocaleTimeString('en-US', {
  1029→                              hour: '2-digit',
  1030→                              minute: '2-digit',
  1031→                              second: '2-digit',
  1032→                            })}
  1033→                          </TableCell>
  1034→                        </TableRow>
  1035→                      )
  1036→                    })}
  1037→                  </TableBody>
  1038→                </Table>
  1039→              </div>
  1040→            )}
  1041→          </CardContent>
  1042→        </Card>
  1043→      </motion.div>
  1044→
  1045→      {/* Complete Session Dialog */}
  1046→      <Dialog open={showCompleteDialog} onOpenChange={setShowCompleteDialog}>
  1047→        <DialogContent>
  1048→          <DialogHeader>
  1049→            <DialogTitle>Complete Inventory Session?</DialogTitle>
  1050→            <DialogDescription>
  1051→              This will finalize the session &ldquo;{session.name}&rdquo;. You will
  1052→              not be able to scan additional items after completion.
  1053→              {session.totalExpected > 0 && session.totalScanned < session.totalExpected && (
  1054→                <span className="block mt-2 text-amber-600 font-medium">
  1055→                  <AlertTriangle className="h-4 w-4 inline mr-1" />
  1056→                  Warning: {session.totalExpected - session.totalScanned} expected items have not been scanned yet.
  1057→                </span>
  1058→              )}
  1059→            </DialogDescription>
  1060→          </DialogHeader>
  1061→          <DialogFooter>
  1062→            <Button
  1063→              variant="outline"
  1064→              onClick={() => setShowCompleteDialog(false)}
  1065→              disabled={completing}
  1066→            >
  1067→              Cancel
  1068→            </Button>
  1069→            <Button onClick={handleComplete} disabled={completing}>
  1070→              {completing ? 'Completing...' : 'Complete Session'}
  1071→            </Button>
  1072→          </DialogFooter>
  1073→        </DialogContent>
  1074→      </Dialog>
  1075→    </div>
  1076→  )
  1077→}
