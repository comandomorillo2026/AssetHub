    65→    65→function AnimatedSection({ children, className = '' }: { children: React.ReactNode; className?: string }) {
    66→    66→  const ref = useRef<HTMLDivElement>(null)
    67→    67→  const isInView = useInView(ref, { once: true, margin: '-80px 0px' })
    68→    68→  return (
    69→    69→    <motion.div
    70→    70→      ref={ref}
    71→    71→      initial="hidden"
    72→    72→      animate={isInView ? 'visible' : 'hidden'}
    73→    73→      className={className}
    74→    74→    >
    75→    75→      {children}
    76→    76→    </motion.div>
    77→    77→  )
    78→    78→}
