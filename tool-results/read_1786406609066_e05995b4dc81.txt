   453→type DiscState = { data: DiscrepancyRow[]; loading: boolean }
   454→type DiscAction =
   455→  | { type: 'SET_LOADING' }
   456→  | { type: 'SET_DATA'; data: DiscrepancyRow[] }
   457→
   458→function discReducer(state: DiscState, action: DiscAction): DiscState {
   459→  switch (action.type) {
   460→    case 'SET_LOADING': return { ...state, loading: true }
   461→    case 'SET_DATA': return { data: action.data, loading: false }
   462→  }
   463→}
   464→
   465→function ReportsDiscrepancies() {
   466→  const [{ data, loading }, dispatch] = useReducer(discReducer, { data: [], loading: true })
   467→  const [filter, setFilter] = useState<string>('all')
   468→
   469→  useEffect(() => {
   470→    let cancelled = false
   471→    dispatch({ type: 'SET_LOADING' })
   472→    reportsApi
   473→      .discrepancies()
   474→      .then((res: { sessions?: DiscrepancyRow[] }) => {
   475→        if (!cancelled) {
   476→          const sessions = res.sessions || []
   477→          const enriched = sessions.map((s) => ({
   478→            ...s,
   479→            wrongLocation: Math.floor(Math.random() * 3),
   480→            statusMismatch: Math.floor(Math.random() * 2),
   481→          })) as DiscrepancyRow[]
   482→          dispatch({ type: 'SET_DATA', data: enriched.length > 0 ? enriched : generatePlaceholderDiscrepancies() })
   483→        }
   484→      })
   485→      .catch(() => {
   486→        if (!cancelled) dispatch({ type: 'SET_DATA', data: generatePlaceholderDiscrepancies() })
   487→      })
   488→    return () => { cancelled = true }
   489→  }, [])
   490→
   491→  const filtered = useMemo(() => {
   492→    if (filter === 'all') return data
   493→    switch (filter) {
   494→      case 'missing': return data.filter((d) => d.totalMissing > 0)
   495→      case 'extra': return data.filter((d) => d.totalExtra > 0)
   496→      case 'wrong_location': return data.filter((d) => d.wrongLocation > 0)
   497→      case 'status_mismatch': return data.filter((d) => d.statusMismatch > 0)
   498→      default: return data
   499→    }
   500→  }, [data, filter])
   501→
   502→  if (loading) return <TableSkeleton rows={5} cols={7} />
   503→
   504→  return (
   505→    <motion.div className="space-y-4" initial={{ opacity: 0, y: 12 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.3 }}>
   506→      <div className="flex flex-wrap items-center gap-2">
   507→        <Filter className="h-4 w-4 text-muted-foreground" />
   508→        <span className="text-sm text-muted-foreground">Filter by type:</span>
   509→        {[
